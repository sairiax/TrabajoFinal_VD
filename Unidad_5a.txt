Introducción a Streamlit
 En resumen, el modelo de ejecución es:

 Las aplicaciones Streamlit son scripts de Python que se

ejecutan de arriba a abajo
 Cada vez que un usuario abre una pestaña del navegador que
apunta a su aplicación, el script se vuelve a ejecutar
 A medida que se ejecuta el script, Streamlit dibuja su salida en
vivo en un navegador
 Los scripts usan la memoria caché para evitar volver a calcular
funciones costosas o cargar archivos grandes
 Cada vez que un usuario interactúa con un widget, su secuencia
de comandos se vuelve a ejecutar y el valor de salida de ese
widget se establece en el nuevo valor durante esa ejecución
 Las aplicaciones Streamlit pueden contener varias páginas, que
se definen en archivos .py separados en una carpeta de páginas

def display_map(df, year, quarter, sex):
df = df[(df['Año'] == year) & (df['Trimestre'] == quarter) & (df['Sexo'] == sex)]
m = folium.Map(location=[40.42, -3.7], zoom_start=5, control_scale=True)
coropletas = folium.Choropleth(geo_data=prov_geo,name="choropleth",data=df,
columns=("codigo", "Paro"),key_on="properties.codigo", fill_color='YlGn’,
fill_opacity=0.7,line_opacity=1.0,legend_name="Tasa de Paro (%)")
coropletas.geojson.add_to(m)
for feature in coropletas.geojson.data['features']:
code = feature['properties']['codigo']
feature['properties']['Provincia'] = prov_dict[code]
coropletas.geojson.add_child(folium.features.GeoJsonTooltip(['Provincia'], labels=False))
folium.LayerControl().add_to(m)
st_map = st_folium(m, width=700, height=450)
codigo = '00'
if st_map['last_active_drawing']:
codigo = st_map['last_active_drawing']['properties']['codigo']
return codigo

def grafica_paro(df, year, quarter, prov_name, prov_code, sex):
df = df[(df['codigo'] == prov_code) &
(df['Sexo'] == sex)].reset_index()
idx = (df.index[(df['Año'] == year) &
(df['Trimestre'] == quarter)].tolist())[0]
df = df.iloc[idx:min(len(df), idx+12)]
df = df.sort_values(by=['Año', 'Trimestre'], ascending=True)
df['Periodo'] = df['Año'].astype(str) + ' T’ +
df['Trimestre'].astype(str)
if '/' in prov_name:
prov_name = prov_name.split('/')[0]
fig = px.bar(df, x='Periodo', y='Paro', title=f'{prov_name} - {sex}')
fig.update_layout(yaxis=dict(range=[df['Paro'].min() - 1,
df['Paro'].max() + 1]))
st.plotly_chart(fig)
